path_num <- paste0(path_data, "rasters_coarse_same_extent_and_res/numeric/")

# Select rasters that are useful for the targeted species
pred_to_keep <- c("bio01", "bio02", "bio03", "bio04", "bio05", "bio06", "bio07", "bio08", "bio09", "bio10", "bio11", "bio12", "bio13", "bio14", "bio15", "bio16", "bio17", "bio18", "bio19",

                  "coverfraction_crops", "coverfraction_grass", "coverfraction_shrub", "coverfraction_tree", 
                  "ECM_plant_div", "ECM_plant_occ", "elevation", "GHM", "HII", "soil_bdod", 
                  "soil_cec", "soil_clay", "soil_nitrogen", "soil_phh2o", "soil_sand", "soil_silt", "soil_soc", "species_div")
pred_selected <- pred[[names(pred) %in% pred_to_keep]]
var_numeric <- names(pred_selected)

# Match CRS and extract training data at points
plots_ll <- sf::st_transform(plots, terra::crs(pred_selected))
plots_v <- terra::vect(plots_ll)
Xtrain <- terra::extract(pred_selected, plots_v)
Xtrain <- Xtrain[, -1, drop = FALSE]

# Keep only complete rows for all predictors used
Xtrain <- Xtrain[stats::complete.cases(Xtrain[, vars_numeric, drop = FALSE]), vars_numeric, drop = FALSE]

# Numeric subset for filtering
Xnum <- Xtrain[, var_numeric, drop = FALSE]

# Remove zero-variance numeric predictors
sd_vec <- apply(Xnum, 2, stats::sd, na.rm = TRUE)
zero_sd <- names(sd_vec)[is.na(sd_vec) | sd_vec == 0]
Xnum0 <- Xnum[, !names(Xnum) %in% zero_sd, drop = FALSE]

# Correlation filter
cmat <- stats::cor(Xnum0, use = "pairwise.complete.obs", method = "pearson")
high_pairs <- which(abs(cmat) > 0.7 & upper.tri(cmat), arr.ind = TRUE)
drop_cor <- character(0)
if(nrow(high_pairs) > 0) drop_cor <- unique(colnames(cmat)[high_pairs[, 2]])
Xnum1 <- Xnum0[, setdiff(colnames(Xnum0), drop_cor), drop = FALSE]

# VIF filter
v <- usdm::vifstep(Xnum1, th = 10)
keep_vif <- v@results$Variables
Xnum2 <- Xnum1[, keep_vif, drop = FALSE]

# Final predictor raster set
pred_sel <- pred_selected[[colnames(Xnum2)]]

# Mask to land
pred_land <- terra::mask(pred_sel, land_v)
pred_land <- terra::crop(pred_land, land_v)
pred_land <- terra::trim(pred_land)

# Export final predictors

######### Calculate area of applicability and dissimilarity index ############################
##############################################################################################

# Load scaled predictors as a stack

pred_aoa_sc <- terra::rast(files_scaled)

# Numeric predictors used for AOA
vars_aoa <- colnames(Xnum2)
# Reorder explicitly to match training table
pred_aoa_sc <- pred_aoa_sc[[vars_aoa]]

# Scaling parameters from training data
mu <- sapply(Xnum2, mean, na.rm = TRUE)
sdv <- sapply(Xnum2, stats::sd, na.rm = TRUE)

# Paths
path_scaled <- paste0(path_geopackages, "predictors_scaled/")
dir.create(path_scaled, showWarnings = FALSE, recursive = TRUE)

# Scale and export one by one
for(v in vars_aoa) {
  r <- pred_land[[v]]
  r <- (r - mu[v]) / sdv[v]
  names(r) <- v
  terra::writeRaster(r, filename = paste0(path_scaled, v, ".tif"), overwrite = TRUE)
  rm(r)
  gc()}

# Reload scaled predictors as a stack
files_scaled <- list.files(path_scaled, pattern = "\\.tif$", full.names = TRUE)
pred_aoa_sc <- terra::rast(files_scaled)

# Reorder explicitly to match training table
pred_aoa_sc <- pred_aoa_sc[[vars_aoa]]

# Scale training table with same parameters
Xnum2_sc <- as.data.frame(scale(Xnum2, center = mu, scale = sdv))

# Check order
identical(names(pred_aoa_sc), colnames(Xnum2_sc))

# Calculate AOA
aoa_res <- CAST::aoa(newdata = pred_aoa_sc, train = Xnum2_sc, verbose = TRUE)

# Derive DI
DI <- aoa_res$DI
Tvalue <- as.numeric(aoa_res$parameters$threshold)
AOA <- aoa_res$AOA

# Mask / crop / trim outputs to land
DI_land <- terra::mask(DI, land_v)
DI_land <- terra::crop(DI_land, land_v)
DI_land <- terra::trim(DI_land)
AOA_land <- terra::mask(AOA, land_v)
AOA_land <- terra::crop(AOA_land, land_v)
AOA_land <- terra::trim(AOA_land)

# Export
terra::writeRaster(DI_land, paste0(path_outputs, "DI_raw.tif"), overwrite = TRUE)
terra::writeRaster(AOA_land, paste0(path_outputs, "AOA_raw.tif"), overwrite = TRUE)

##############################################################################################
############# Post-processing and classifying dissimilarity index ############################
##############################################################################################

# Treat with empty cells
alt <- pred_land[["elevation"]]
alt[alt>=-9999] <- 1
alt <- crop(alt, DI_land)
b <- DI_land
b[b>=-9999] <- 0
b[is.na(b)] <- 1
ab <- alt * b
x <- DI_land
x[is.na(x)] <-0
ab[ab>=0]<-1
DI_land_corr <- x * ab

# Export
terra::writeRaster(DI_land_corr, paste0(path_outputs, "DI.tif"), overwrite = TRUE)

# Classify DI into novelty bands
novelty <- terra::classify(DI_land_corr, rbind(c(-Inf, 0.00001, 0), c(0.00001, Tvalue, 1), c(Tvalue, 2 * Tvalue, 2), c(2 * Tvalue, Inf, 3)), include.lowest = TRUE, right = TRUE)

    "0, no data",

    "T-2T, slightly undersampled",
    "> 2T, moderately to highly undersampled"))
novelty <- terra::as.factor(novelty)
levels(novelty) <- levels_df

# Summary statistics on novelty raster

f2$count[is.na(f2$count)] <- 0

pr <- sf::st_read(paste0(path_data, "vectors/GlobalPlantDiversity (Lirong Cai ea 2023)/pr_Ensemble_Prediction_7774_LongLat/pr_Ensemble_Prediction_7774_LongLat.shp"))

# Turn off spherical geometry
sf::sf_use_s2(FALSE)
# Define the number of bins for host richness on the x-axis and latitude on the y-axis
nbx <- 30
nby <- 30
# Make the host-richness polygons and plot points geometrically valid before spatial operations
sr <- sf::st_make_valid(sr)
pr <- sf::st_make_valid(pr)
plots <- sf::st_make_valid(plots)

# ================= Plant species richness and latitude coverage ==================

# Build background table: plant species richness × latitude
sr$area_hex <- as.numeric(sf::st_area(sf::st_transform(sr, 6933)))
sr_cent <- sf::st_transform(sf::st_centroid(sf::st_transform(sr, 6933)), 4326)
cent_xy <- sf::st_coordinates(sr_cent)
sr_lat <- sr %>% sf::st_drop_geometry() %>% dplyr::transmute(value_sr = as.numeric(value), area_hex = area_hex, lat = cent_xy[,2]) %>% dplyr::filter(is.finite(value_sr), is.finite(area_hex), is.finite(lat))

# Assign plant species richness to sampled plots
plots_sr <- plots %>% dplyr::select(-dplyr::any_of("value_sr"))
if (sf::st_crs(plots_sr) != sf::st_crs(sr)) plots_sr <- sf::st_transform(plots_sr, sf::st_crs(sr))
plots_sr <- sf::st_join(plots_sr, sr %>% dplyr::select(value_sr = value), join = sf::st_within, left = TRUE)
pts <- plots_sr %>% sf::st_drop_geometry() %>% as.data.frame() %>% dplyr::mutate(value_sr = as.numeric(value_sr), Latitude = as.numeric(Latitude)) %>% dplyr::filter(is.finite(value_sr), is.finite(Latitude))

# Define richness × latitude bins
x_breaks <- seq(min(sr_lat$value_sr, na.rm = TRUE), max(sr_lat$value_sr, na.rm = TRUE), length.out = nbx + 1)
y_breaks <- seq(min(sr_lat$lat, na.rm = TRUE), max(sr_lat$lat, na.rm = TRUE), length.out = nby + 1)
x_mid <- (head(x_breaks, -1) + tail(x_breaks, -1)) / 2
y_mid <- (head(y_breaks, -1) + tail(y_breaks, -1)) / 2

# Summarise global background area by bin
sr_lat <- sr_lat %>% dplyr::mutate(x_bin = cut(value_sr, breaks = x_breaks, include.lowest = TRUE, right = TRUE), y_bin = cut(lat, breaks = y_breaks, include.lowest = TRUE, right = TRUE))
bg <- sr_lat %>% dplyr::filter(!is.na(x_bin), !is.na(y_bin)) %>% dplyr::group_by(x_bin, y_bin) %>% dplyr::summarise(area_sum = sum(area_hex, na.rm = TRUE), .groups = "drop") %>% dplyr::mutate(share = area_sum / sum(area_sum, na.rm = TRUE)) %>% dplyr::filter(is.finite(share), share > 0) %>% as.data.frame()
bg$x <- x_mid[as.integer(bg$x_bin)]
bg$y <- y_mid[as.integer(bg$y_bin)]

# Identify bins occupied by sampled plots
pts <- pts %>% dplyr::mutate(x_bin = cut(value_sr, breaks = x_breaks, include.lowest = TRUE, right = TRUE), y_bin = cut(Latitude, breaks = y_breaks, include.lowest = TRUE, right = TRUE))
dots <- pts %>% dplyr::filter(!is.na(x_bin), !is.na(y_bin)) %>% dplyr::distinct(x_bin, y_bin) %>% as.data.frame()
dots$x <- x_mid[as.integer(dots$x_bin)]
dots$y <- y_mid[as.integer(dots$y_bin)]

# Plot the global background area share as tiles and overlay sampled bins as white dots
b <- ggplot2::ggplot(bg, ggplot2::aes(x = x, y = y)) +
  ggplot2::geom_tile(ggplot2::aes(fill = share)) +
  ggplot2::geom_point(data = dots, ggplot2::aes(x = x, y = y), color = "white", size = 1.5) +
  ggplot2::scale_fill_viridis_c(trans = "log10", option = "viridis", labels = scales::percent_format(accuracy = 0.01), name = "Global area share") +
  ggplot2::labs(x = "Plant species richness", y = "Latitude (°)") +
  ggplot2::coord_cartesian(expand = FALSE) +
  ggplot2::theme_classic()
b

# Export the host-latitude bin figure.
setwd(path_figures)
ggplot2::ggsave("HOST-LAT BIN.png", plot = b, width = 14, height = 11, units = "cm", dpi = 600, bg = "white")
ggplot2::ggsave("HOST-LAT BIN.tif", plot = b, width = 14, height = 11, units = "cm", dpi = 600, compression = "lzw", bg = "white")
ggplot2::ggsave("HOST-LAT BIN.pdf", plot = b, width = 14, height = 11, units = "cm", device = cairo_pdf, bg = "white")

# Count how many bins are occupied by at least one plot
n_bins_total <- nrow(bg)
n_bins_covered <- nrow(dots)
prop_bins_covered <- n_bins_covered / n_bins_total
cat("Covered bins:", n_bins_covered, "of", n_bins_total, "\n")
cat("Proportion of bins covered:", round(prop_bins_covered * 100, 1), "%\n")

# Calculate how much global area is represented by the occupied hbins
dots_bg <- dots %>% dplyr::left_join(bg %>% dplyr::select(x_bin, y_bin, area_sum, share), by = c("x_bin", "y_bin"))
area_covered <- sum(dots_bg$area_sum, na.rm = TRUE)
area_total <- sum(bg$area_sum, na.rm = TRUE)
prop_area_covered <- area_covered / area_total
cat("Covered global area:", round(prop_area_covered * 100, 1), "%\n")

# Build global background table for plant species richness × phylogenetic richness
sr_bg$area_hex <- as.numeric(sf::st_area(sf::st_transform(sr_bg, 6933)))

sr_cent$area_hex <- sr_bg$area_hex

rich_bg <- rich_bg_sf %>% sf::st_drop_geometry() %>% as.data.frame() %>% dplyr::mutate(value_sr = as.numeric(value_sr), value_pr = as.numeric(value_pr), area_hex = as.numeric(area_hex)) %>% dplyr::filter(is.finite(value_sr), is.finite(value_pr), is.finite(area_hex))

# Assign plant diversity values to sampled plots
plots_rich <- plots %>% dplyr::select(-dplyr::any_of(c("value_sr", "value_pr")))
if (sf::st_crs(plots_rich) != sf::st_crs(sr_bg)) plots_rich <- sf::st_transform(plots_rich, sf::st_crs(sr_bg))
plots_rich <- sf::st_join(plots_rich, sr_bg %>% dplyr::select(value_sr = value), join = sf::st_within, left = TRUE)
plots_rich <- sf::st_join(plots_rich, pr_bg %>% dplyr::select(value_pr = value), join = sf::st_within, left = TRUE)
pts <- plots_rich %>% sf::st_drop_geometry() %>% as.data.frame() %>% dplyr::mutate(value_sr = as.numeric(value_sr), value_pr = as.numeric(value_pr)) %>% dplyr::filter(is.finite(value_sr), is.finite(value_pr))

# Define two-dimensional bin structure
x_breaks <- seq(min(rich_bg$value_sr, na.rm = TRUE), max(rich_bg$value_sr, na.rm = TRUE), length.out = nbx + 1)
y_breaks <- seq(min(rich_bg$value_pr, na.rm = TRUE), max(rich_bg$value_pr, na.rm = TRUE), length.out = nby + 1)
x_mid <- (head(x_breaks, -1) + tail(x_breaks, -1)) / 2
y_mid <- (head(y_breaks, -1) + tail(y_breaks, -1)) / 2

# # Summarise global background area by diversity bin
rich_bg <- rich_bg %>% dplyr::mutate(x_bin = cut(value_sr, breaks = x_breaks, include.lowest = TRUE, right = TRUE, labels = FALSE), y_bin = cut(value_pr, breaks = y_breaks, include.lowest = TRUE, right = TRUE, labels = FALSE))
bg <- rich_bg %>% dplyr::filter(!is.na(x_bin), !is.na(y_bin)) %>% dplyr::group_by(x_bin, y_bin) %>% dplyr::summarise(area_sum = sum(area_hex, na.rm = TRUE), .groups = "drop") %>% dplyr::mutate(share = area_sum / sum(area_sum, na.rm = TRUE), x = x_mid[x_bin], y = y_mid[y_bin]) %>% dplyr::filter(is.finite(share), share > 0) %>% as.data.frame()

# Identify diversity bins occupied by sampled plots
pts <- pts %>% dplyr::mutate(x_bin = cut(value_sr, breaks = x_breaks, include.lowest = TRUE, right = TRUE, labels = FALSE), y_bin = cut(value_pr, breaks = y_breaks, include.lowest = TRUE, right = TRUE, labels = FALSE))
dots <- pts %>% dplyr::filter(!is.na(x_bin), !is.na(y_bin)) %>% dplyr::distinct(x_bin, y_bin) %>% dplyr::mutate(x = x_mid[x_bin], y = y_mid[y_bin]) %>% as.data.frame()

# Plot
b <- ggplot2::ggplot(bg, ggplot2::aes(x = x, y = y)) +
  ggplot2::geom_tile(ggplot2::aes(fill = share), width = diff(x_breaks)[1], height = diff(y_breaks)[1]) +
  ggplot2::geom_point(data = dots, ggplot2::aes(x = x, y = y), color = "white", size = 1.5) +

# Prepare projected countries and plot points
sf::sf_use_s2(FALSE)
if (!exists("crs_robin")) crs_robin <- "+proj=robin +datum=WGS84 +units=m +no_defs"
countries_m <- countries %>% sf::st_make_valid() %>% sf::st_transform(crs_robin)
plot_sf_m <- plots %>% sf::st_make_valid() %>% sf::st_transform(crs_robin)
if (!("N_Samples" %in% names(plot_sf_m))) plot_sf_m$N_Samples <- 1

# Simplify duplicated plot locations and sum samples per location
xy0 <- sf::st_coordinates(plot_sf_m)
plot_sf_m_simple <- plot_sf_m %>%
  sf::st_drop_geometry() %>%
  dplyr::mutate(x = xy0[,1], y = xy0[,2], n_samples = as.numeric(N_Samples)) %>%
  dplyr::filter(is.finite(x), is.finite(y), is.finite(n_samples)) %>%
  dplyr::group_by(x, y) %>%
  dplyr::summarise(n_samples = sum(n_samples, na.rm = TRUE), .groups = "drop") %>%
  sf::st_as_sf(coords = c("x", "y"), crs = sf::st_crs(countries_m), remove = FALSE)

# Build spatstat point pattern in projected metres
xy <- sf::st_coordinates(plot_sf_m_simple)
bb <- sf::st_bbox(countries_m)
W <- spatstat.geom::owin(xrange = c(bb["xmin"], bb["xmax"]), yrange = c(bb["ymin"], bb["ymax"]))
pp <- spatstat.geom::ppp(x = xy[,1], y = xy[,2], window = W)

# Create weighted kernel density image; sigma is in metres because data are now projected
dens_im <- spatstat.explore::density.ppp(pp, sigma = 500000, weights = plot_sf_m_simple$n_samples, edge = TRUE, dimyx = 900)

# Convert density image to raster and assign CRS
dens_stars <- stars::st_as_stars(dens_im)
dens_rast <- terra::rast(dens_stars)
land_vect <- terra::vect(countries_m)
terra::crs(dens_rast) <- terra::crs(land_vect)

# Mask density to land
dens_land <- terra::mask(dens_rast, land_vect)
                         
# Convert density raster to dataframe
dens_df <- terra::as.data.frame(dens_land, xy = TRUE, na.rm = TRUE)
names(dens_df)[3] <- "density"
dens_df <- dens_df %>% dplyr::filter(is.finite(density))

# Normalize density with 99% cap to avoid saturation by extreme hotspot cells
dens_min <- min(dens_df$density, na.rm = TRUE)
dens_cap <- stats::quantile(dens_df$density, 0.99, na.rm = TRUE)
dens_df <- dens_df %>% dplyr::mutate(density_scaled_mod = (pmin(density, dens_cap) - dens_min) / (dens_cap - dens_min), density_scaled_mod = pmin(pmax(density_scaled_mod, 0), 1))

# Optional: set very weak density values to transparent. Use 0 to keep all land density visible
density_floor_prop <- 0
dens_df <- dens_df %>% dplyr::mutate(density_scaled_mod = dplyr::if_else(density_scaled_mod <= density_floor_prop, NA_real_, density_scaled_mod))

# Get the projected y-coordinate for latitude -54 degrees to cut Antarctica
lat_line <- sf::st_sfc(sf::st_linestring(matrix(c(-180, -54, 180, -54), ncol = 2, byrow = TRUE)), crs = 4326)
lat_line_robin <- sf::st_transform(lat_line, crs_robin)
y_cut <- sf::st_coordinates(lat_line_robin)[1, "Y"]

# Plot density map
p <- ggplot2::ggplot() +
  ggplot2::geom_tile(data = dens_df, ggplot2::aes(x = x, y = y, fill = density_scaled_mod), alpha = 0.9) +
  ggplot2::geom_sf(data = countries_m, fill = NA, color = "grey35", linewidth = 0.15) +
  ggplot2::scale_fill_viridis_c(option = "mako", direction = -1, name = "Normalized sample density", limits = c(0, 1), breaks = c(0, 0.25, 0.50, 0.75, 1), labels = c("0.00", "0.25", "0.50", "0.75", "1.00"), na.value = "transparent") +
  ggplot2::coord_sf(xlim = c(bb["xmin"], bb["xmax"]), ylim = c(y_cut, bb["ymax"]), expand = FALSE) +
  ggplot2::guides(fill = ggplot2::guide_colorbar(barwidth = 24, barheight = 0.6, title.position = "top", label.position = "bottom")) +
  ggplot2::theme_minimal(base_size = 12) +
  ggplot2::theme(axis.title = ggplot2::element_blank(), axis.text = ggplot2::element_blank(), axis.ticks = ggplot2::element_blank(), panel.grid = ggplot2::element_blank(), legend.position = "bottom", legend.title.align = 0.5)

setwd(path_figures)

ggplot2::ggsave("PLOTS DENSITY.tif", plot = p, width = 25, height = 14, units = "cm", dpi = 600, compression = "lzw", bg = "white")
ggplot2::ggsave("PLOTS DENSITY.pdf", plot = p, width = 25, height = 14, units = "cm", device = cairo_pdf, bg = "white")
