# =============================================================================
# Format BLAST results and merge taxonomic assignments with the FunLeaf OTU table          
# In HPC 
# =============================================================================
import os
import glob
import pandas as pd

# Path to BLAST result files
blast_results_dir = "<PROJECT_ROOT>/Step2_Results/05.LULU/BLAST_results"
blast_result_files = sorted(glob.glob(os.path.join(blast_results_dir, "OTUs_LULU.part_*.m8")))

# Custom header
formatted_blast_columns = [
    "OTUid", "accession no", "Kingdom", "phylum", "class", "order", "family",
    "genus", "species", "ECM status", "...11", "SeqIdentity", "AlignLen",
    "MismatchN", "GapOpenings", "QueryStart", "QueryEnd", "TargetStart",
    "TargetEnd", "Evalue", "BitScore"
]

formatted_blast_tables = []

# Read and process each file
for blast_result_file in blast_result_files:
    df = pd.read_csv(blast_result_file, sep="\t", header=None)
    df.columns = [
        "OTUid", "subject", "SeqIdentity", "AlignLen", "MismatchN", "GapOpenings",
        "QueryStart", "QueryEnd", "TargetStart", "TargetEnd", "Evalue", "BitScore"
    ]

    # Split taxonomy field
    taxonomy_split = df["subject"].str.split(";", expand=True)

    # Ensure there are enough taxonomy columns
    max_cols = 10  # accession no to ...11
    for i in range(max_cols - taxonomy_split.shape[1]):
        taxonomy_split[taxonomy_split.shape[1] + i] = None

    taxonomy_split.columns = [
        "accession no", "Kingdom", "phylum", "class", "order", "family",
        "genus", "species", "ECM status", "...11"
    ]

    # Merge with original blast info
    final_df = pd.concat([df["OTUid"], taxonomy_split, df.iloc[:, 2:]], axis=1)
    formatted_blast_tables.append(final_df)

# Concatenate all dataframes
merged_df = pd.concat(formatted_blast_tables, ignore_index=True)
merged_df.columns = formatted_blast_columns

# Save the final file of blast results
output_path = os.path.join(blast_results_dir, "all_otus_blast_formatted.tsv")
merged_df.to_csv(output_path, sep="\t", index=False)
print(f"Saved merged file to: {output_path}")

# Apply Evalue-based filters

# Verify all rows are accounted for
all_evalue_groups = pd.concat([low_confidence_evalue, intermediate_confidence_evalue, high_confidence_evalue])
unclassified_evalue_rows = df.loc[~df.index.isin(all_evalue_groups.index)]

print(f"Total rows: {len(df)}")
print(f"Low confidence: {len(low_confidence_evalue)}, intermediate confidence: {len(intermediate_confidence_evalue)}, high confidence: {len(high_confidence_evalue)}")
print(f"Unaccounted rows (should be 0): {len(unclassified_evalue_rows)}")

# Output directory
output_dir = "<PROJECT_ROOT>/Step2_Results/05.LULU"

# Save each filtered group as separate CSV files

# Step 2: Add empty taxonomy columns to low-confidence assignments
taxonomy_columns = ['Kingdom', 'phylum', 'class', 'order', 'family', 'genus', 'species']
for col in taxonomy_columns:
    low_confidence_assignments[col] = pd.NA  # adds column if not already present

# Step 3: Combine dataframes
combined = pd.concat([low_confidence_assignments, high_confidence_assignments, intermediate_confidence_assignments], ignore_index=True)

# Ensure SeqIdentity is numeric
combined['SeqIdentity'] = pd.to_numeric(combined['SeqIdentity'], errors='coerce')

# Step 4: Apply identity-based filtering
combined.loc[combined['SeqIdentity'] < 75, ['class', 'order', 'family', 'genus', 'species']] = pd.NA
combined.loc[(combined['SeqIdentity'] >= 75) & (combined['SeqIdentity'] < 80), ['order', 'family', 'genus', 'species']] = pd.NA
combined.loc[(combined['SeqIdentity'] >= 80) & (combined['SeqIdentity'] < 85), ['family', 'genus', 'species']] = pd.NA
combined.loc[(combined['SeqIdentity'] >= 85) & (combined['SeqIdentity'] < 90), ['genus', 'species']] = pd.NA

# Step 5: Replace NA and "." with "Unknown" in taxonomy columns
for col in taxonomy_columns:
    combined[col] = combined[col].replace(["NA", ".", None, pd.NA], "Unknown")
    combined[col] = combined[col].fillna("Unknown")

# Step 6: Save to CSV
combined.to_csv(output_path, index=False)
print(f"✅ Cleaned file saved to: {output_path}")

# We applied a threshold of 250 bp to remove very short OTU sequences, which were assumed to be artefactual. After selecting

# =============================================================================
#!/bin/bash

INPUT="subset_all_OTUs.fa"     # or your main FASTA
OUTPUT="otus_shorter_than_250bp.fa"

awk -v maxlen=250 '
/^>/ {
    # print previous sequence if condition met
    if (seqlen > 0 && seqlen < maxlen) {
        print header
        print seq
    }
    header = $0
    seq = ""
    seqlen = 0
    next
}

{
    seq = seq $0
    seqlen += length($0)
}

END {
    # last sequence
    if (seqlen > 0 && seqlen < maxlen) {
        print header
        print seq
    }
}
' "$INPUT" > "$OUTPUT"

echo "✅ Done"

echo -n "Sequences < 250 bp: "
grep -c "^>" "$OUTPUT"

# =============================================================================
# calculate OTU richness (number of unique OTUs) and total reads per fungal taxonomic group
# =============================================================================

# 1. Subset the OTU table (reads) and taxonomic info
# Assuming OTU counts are columns 2 to 6750 (FL codes), and taxonomic info from column 426
otu_table <- All_OTU_with_FL[, seq.int(2, ncol(All_OTU_with_FL) - 22), drop = FALSE]  # skip OTU name column (1), read counts only
taxonomy_info <- All_OTU_with_FL[, c("OTU", "Kingdom", "phylum", "class", "order", "family", "genus")]

# Add total read count per OTU
taxonomy_info$TotalReads <- rowSums(otu_table)

# 2. Function to calculate richness and read counts per taxonomic level
get_richness_reads <- function(level) {
  aggregate_data <- aggregate(TotalReads ~ get(level), data = taxonomy_info, FUN = sum)
  richness_data <- aggregate(OTU ~ get(level), data = taxonomy_info, FUN = function(x) length(unique(x)))
  result <- merge(aggregate_data, richness_data, by = "get(level)")
  colnames(result) <- c(level, "Total_Reads", "OTU_Richness")
  return(result)
}

# 3. Apply function to each level
Kingdom_summary <- get_richness_reads("Kingdom")
phylum_summary <- get_richness_reads("phylum")
class_summary  <- get_richness_reads("class")
order_summary  <- get_richness_reads("order")
family_summary <- get_richness_reads("family")
genus_summary  <- get_richness_reads("genus")

## remove samples with the above thresholds (IN R)
# =============================================================================
####################################################################################################################
# First keep FunLeaf samples (Bioproject includes secuences from samples of other projects)
# list of FunLeaf samples are under column 'FunLeaf_ID' in the sample metadata acompanying the Raw Sequences in ENA 
#####################################################################################################################

keep_list <- read.csv("<PROJECT_ROOT>/samples_to_keep_after_threshold.csv",
                      check.names = FALSE, stringsAsFactors = FALSE)

if (!"FunLeaf_ID" %in% names(keep_list)) {
  stop("The sample keep-list must contain a 'FunLeaf_ID' column.")
}
keep_ids <- trimws(as.character(keep_list$FunLeaf_ID))
keep_ids <- keep_ids[!is.na(keep_ids) & nzchar(keep_ids)]
if (anyDuplicated(keep_ids)) stop("The sample keep-list contains duplicate FL codes.")

metadata_count <- 21
sample_cols <- seq.int(2, ncol(All_OTU_with_FL) - metadata_count)
available_samples <- colnames(All_OTU_with_FL)[sample_cols]
missing_samples <- setdiff(keep_ids, available_samples)
if (length(missing_samples) > 0) {
  stop("Requested FL codes not found: ", paste(missing_samples, collapse = ", "))
}
keep_cols <- match(keep_ids, available_samples) + 1
metadata_cols <- seq.int(ncol(All_OTU_with_FL) - metadata_count + 1,
                         ncol(All_OTU_with_FL))
Data_clean <- All_OTU_with_FL[, c(1, keep_cols, metadata_cols), drop = FALSE]
cat("Samples retained from threshold keep-list:", length(keep_ids), "\n")

# 1. Identify sample read columns
otu_data <- Data_clean

# Exclude OTU ID (1st column) and last 21 columns (metadata)
read_cols <- seq.int(2, ncol(otu_data) - 21)

# 2. Calculate how many samples each OTU appears in (non-zero counts)
otu_counts <- rowSums(otu_data[, read_cols] > 0, na.rm = TRUE)

# 3. Keep only rows (OTUs) that appear in more than one sample
Data_clean <- otu_data[otu_counts > 1, ]

# =============================================================================
# Export FILE1: All OTUs (all kingdoms) AFTER thresholds  (Used in FunLeaf dataset) IN R
# =============================================================================
# Data_clean = dataset after removing unwanted columns and singletons
read_cols_clean <- seq.int(2, ncol(Data_clean) - 21)

All_OTUs_AfterThresholds <- Data_clean %>%
  mutate(Abundance = rowSums(across(all_of(read_cols_clean)), na.rm = TRUE))

# FILE: All OTUs (all kingdoms) AFTER thresholds
write.csv(All_OTUs_AfterThresholds,
          "<PROJECT_ROOT>/all_otus_after_thresholds.csv",
          row.names = FALSE)

# =============================================================================
# Export FILE2: Fungi OTU after thresholds  (Used in FunLeaf dataset) 
# =============================================================================

# Separating Fungi data (including cf.Fungi)
Fungi_data <- Data_clean[grepl("Fungi|cf.Fungi", Data_clean$Kingdom), ]

# 1. Identify sample read columns
otu_data <- Fungi_data

# Exclude OTU ID (1st column) and last 21 columns (metadata)
read_cols <- seq.int(2, ncol(otu_data) - 21)

# 2. Calculate how many samples each OTU appears in (non-zero counts)
otu_counts <- rowSums(otu_data[, read_cols] > 0, na.rm = TRUE)

# 3. Keep only rows (OTUs) that appear in more than one sample
Fungi_no_singletons <- otu_data[otu_counts > 1, ]
Fungi_OTUs_after <- Fungi_no_singletons %>%
  mutate(Abundance = rowSums(across(all_of(read_cols)), na.rm = TRUE))

# FILE 4: Fungi OTUs AFTER thresholds
write.csv(Fungi_OTUs_after,
          "<PROJECT_ROOT>/fungal_otus_after_threshold.csv",
          row.names = FALSE)

# =============================================================================
# Create and Export Taxonomy Table for All Kingdoms from Data_clean                                                      IN R
# =============================================================================

# Identify the sample read columns (exclude metadata)
read_cols_clean <- seq.int(2, ncol(Data_clean) - 21)

# Calculate total abundance (row sum of reads across all samples)
clean_total_abundance <- rowSums(Data_clean[, read_cols_clean], na.rm = TRUE)

# Select taxonomy columns + OTU ID + total abundance
taxonomy_all_kingdoms <- Data_clean[, c("OTU", "Kingdom", "phylum", "class", "order", "family", "genus", "species")]
taxonomy_all_kingdoms$TotalAbundance <- clean_total_abundance

# Save to Excel
write.xlsx(
  taxonomy_all_kingdoms,
  file = "<PROJECT_ROOT>/taxonomy_all_kingdoms.xlsx",
  overwrite = TRUE
)
